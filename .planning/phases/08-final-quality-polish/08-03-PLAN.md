  <action>
    1. Create `PaultUITests/AccessibilityAuditUITests.swift` with the following tests (per research Pattern 1):

       ```swift
       import XCTest

       final class AccessibilityAuditUITests: XCTestCase {
           override func setUp() {
               super.setUp()
               continueAfterFailure = false
           }

           @MainActor
           func testMainWindowAudit() throws {
               let app = XCUIApplication()
               app.launch()
               _ = app.windows.firstMatch.waitForExistence(timeout: 10)
               try app.performAccessibilityAudit { issue in
                   // Suppress dynamicType on ProBadge (fixed size by design)
                   if issue.auditType == .dynamicType,
                      issue.element.identifier == "proBadge" {
                       return true
                   }
                   return false
               }
           }

           @MainActor
           func testBlockEditorAudit() throws {
               let app = XCUIApplication()
               app.launchArguments = ["--screenshot-mode"]
               app.launch()
               _ = app.windows.firstMatch.waitForExistence(timeout: 10)
               // Navigate to block editor -- the screenshot-mode seed data creates prompts with block compositions
               // Use accessibility tree to find and click the first prompt, then switch to Studio tab
               try app.performAccessibilityAudit { issue in
                   if issue.auditType == .dynamicType,
                      issue.element.identifier == "proBadge" {
                       return true
                   }
                   return false
               }
           }
       }
       ```

    2. Run the audits: `xcodebuild test -scheme Pault -destination 'platform=macOS' -only-testing PaultUITests/AccessibilityAuditUITests 2>&1 | xcpretty`

    3. For each audit failure, fix the source view:
       - `.sufficientElementDescription` failures: Add `.accessibilityLabel("descriptive text")` to the element
       - `.contrast` failures: Adjust foreground/background color combinations to meet 4.5:1 ratio
       - `.hitRegion` failures: Ensure clickable elements are at least 44x44 points (use `.frame(minWidth: 44, minHeight: 44)` or padding)
       - Decorative elements that should not be in the accessibility tree: Add `.accessibilityHidden(true)`

    4. Re-run audits until all pass (with only documented suppressions for intentional design choices).

    5. Run existing unit-level accessibility tests to verify no regressions:
       `xcodebuild test -scheme Pault -destination 'platform=macOS' -only-testing PaultTests/AccessibilityTests 2>&1 | xcpretty`

    6. Run keyboard navigation tests:
       `xcodebuild test -scheme Pault -destination 'platform=macOS' -only-testing PaultTests/KeyboardNavigationTests 2>&1 | xcpretty`
  </action>
